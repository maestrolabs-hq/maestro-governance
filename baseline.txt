# Desired state for every repository in the organisation.
#
# One directive per line. Settings apply to every repo listed. Changing a
# value here is the only way to change a repository: the whole point is that
# nothing is configured by hand, because two repositories configured by hand
# drift and nothing notices.
#
# The rulesets themselves are not here yet. They are applied and would need
# the same treatment; this file covers repository settings only, and says so
# rather than implying coverage it does not have.

repo maestro-core
repo maestro-pi-config
repo maestro-governance
repo maestro-project-documentation
repo .github

# Squash only. The ruleset already restricts merge methods; leaving the repo
# settings permissive means the two disagree, and a reader cannot tell which
# one is the intent.
setting allow_squash_merge true
setting allow_merge_commit false
setting allow_rebase_merge false

# A merged branch has served its purpose.
setting delete_branch_on_merge true

# Unused surfaces are attack surface and rot. Issues stay: they are how a
# finding outlives a conversation.
setting has_wiki false
setting has_projects false
setting has_issues true

# ── Organisation ──────────────────────────────────────────────────────────
# These apply to every repository, including ones that do not exist yet, and
# are the reason the estate does not depend on anyone remembering.

org actions.sha_pinning_required true
org actions.default_workflow_permissions read
org actions.can_approve_pull_request_reviews false
org properties.tier.required true
org properties.tier.default_value mature

# Two-factor authentication is required for every member. It is set through
# the web interface only: PATCH /orgs accepts the field and silently leaves it
# unchanged, so this line is checked and never applied.
org two_factor_requirement_enabled true

# ── Rules that must be in effect ──────────────────────────────────────────
# Read from each repository's own branch-rules endpoint, not from the
# organisation ruleset list: listing rulesets needs organisation Administration
# *write*, and an audit token that could rewrite the rules it checks is worth
# stealing. This is also the better assertion -- it proves the rule applies to
# the branch rather than that a ruleset object exists somewhere.
#
# The source matters. A repository-level copy of the floor can be removed by
# whoever owns that repository, which is the whole reason the floor lives on
# the organisation.

rule deletion Organization
rule non_fast_forward Organization
rule pull_request Organization

# ── Per repository ────────────────────────────────────────────────────────
setting web_commit_signoff_required true

# Public, and audited as such. Every security feature this estate relies on --
# code scanning, secret scanning, push protection, dependency review -- is free
# on a public repository and billed per active committer on a private one, and
# Actions minutes are unmetered on public runners and metered on private ones
# at two times for Windows and ten for macOS. The heavy tier runs both weekly.
#
# A repository flipped to private would therefore start billing quietly while
# every other check stayed green. The `visibility-is-frozen` organisation
# ruleset refuses the change outright; this line is the second line of defence,
# because a ruleset can be deleted and an audit notices that a setting moved.
setting visibility public
